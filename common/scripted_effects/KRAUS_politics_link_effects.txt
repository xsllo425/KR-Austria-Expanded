# 大选设立天平后，兑现大选前积累的倾向
KRAUS_bop_apply_pending = {
	while_loop_effect = {
		limit = { check_variable = { KRAUS_bop_pending > 0.04 } }
		add_power_balance_value = { id = KRAUS_regime_balance value = 0.05 }
		subtract_from_variable = { KRAUS_bop_pending = 0.05 }
	}
	while_loop_effect = {
		limit = { check_variable = { KRAUS_bop_pending < -0.04 } }
		add_power_balance_value = { id = KRAUS_regime_balance value = -0.05 }
		add_to_variable = { KRAUS_bop_pending = 0.05 }
	}
}
