	every_country = {
		limit = {
			has_country_flag = SPA_aftermath_of_the_civil_war
			NOT = { has_socialist_government = yes }
		}
		add_opinion_modifier = { target = ROOT modifier = KRAUS_dip_friendship }
		ROOT = { add_opinion_modifier = { target = PREV modifier = KRAUS_dip_friendship } }

KRAUS_acw_contact_effect = {
	set_country_flag = KRAUS_acw_contact_done
	every_country = {
		limit = {
			is_american_tag = yes
			NOT = { has_socialist_government = yes }
			num_of_controlled_states > 5
		}
		add_opinion_modifier = { target = ROOT modifier = KRAUS_dip_friendship }
		ROOT = { add_opinion_modifier = { target = PREV modifier = KRAUS_dip_friendship } }
	}
	if = { limit = { has_country_flag = KRAUS_acw_backed_USA country_exists = USA USA = { NOT = { has_socialist_government = yes } } } add_political_power = 50 }
	if = { limit = { has_country_flag = KRAUS_acw_backed_ACC country_exists = ACC ACC = { NOT = { has_socialist_government = yes } } } add_political_power = 50 }
	if = { limit = { has_country_flag = KRAUS_acw_backed_WCA country_exists = WCA WCA = { NOT = { has_socialist_government = yes } } } add_political_power = 50 }
	if = { limit = { has_country_flag = KRAUS_acw_backed_NEE country_exists = NEE NEE = { NOT = { has_socialist_government = yes } } } add_political_power = 50 }
}
