# 奥地利作用域：已完成的左支国策数（0 到 5），同时算出读条与花费的缩减倍率
KRAUS_pl_recalc = {
	set_variable = { KRAUS_pl_n = 0 }
	if = { limit = { has_completed_focus = KRAUS_pl_L2 } add_to_variable = { KRAUS_pl_n = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L3 } add_to_variable = { KRAUS_pl_n = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L4 } add_to_variable = { KRAUS_pl_n = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L5 } add_to_variable = { KRAUS_pl_n = 1 } }
	# 0.9.1：左支三扇门（L2 / L3 / L3b）与三条第二层（L2b / L3c / L3d）——三选一互斥，一层只走一条
	if = { limit = { has_completed_focus = KRAUS_pl_L3b } add_to_variable = { KRAUS_pl_n = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L2b } add_to_variable = { KRAUS_pl_n = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L3c } add_to_variable = { KRAUS_pl_n = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L3d } add_to_variable = { KRAUS_pl_n = 1 } }
	set_variable = { KRAUS_pl_mult = 1 }
	set_temp_variable = { KRAUS_pl_t = KRAUS_pl_n }
	multiply_temp_variable = { KRAUS_pl_t = 0.1 }
	subtract_from_variable = { KRAUS_pl_mult = KRAUS_pl_t }
	# 整合红利的周期（天）：每完成左支的整合纲要、地块复苏各缩短一档；中支的政治国策各再缩短一档
	set_variable = { KRAUS_div_days = 180 }
	if = { limit = { has_completed_focus = KRAUS_pl_L3 } add_to_variable = { KRAUS_div_days = -30 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L2 } add_to_variable = { KRAUS_div_days = -30 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L3b } add_to_variable = { KRAUS_div_days = -30 } }
	if = { limit = { has_completed_focus = KRAUS_pl_M_diet } add_to_variable = { KRAUS_div_days = -15 } }
	if = { limit = { has_completed_focus = KRAUS_pl_M_lang } add_to_variable = { KRAUS_div_days = -15 } }
	if = { limit = { has_completed_focus = KRAUS_pl_M_seat } add_to_variable = { KRAUS_div_days = -15 } }
	clamp_variable = { var = KRAUS_div_days min = 60 max = 180 }
	set_variable = { KRAUS_div_days_m1 = KRAUS_div_days }
	subtract_from_variable = { KRAUS_div_days_m1 = 1 }
	# 红利的强度：左支第 2 到 5 步各 +1
	set_variable = { KRAUS_div_power = 0 }
	if = { limit = { has_completed_focus = KRAUS_pl_L2 } add_to_variable = { KRAUS_div_power = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L3 } add_to_variable = { KRAUS_div_power = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L4 } add_to_variable = { KRAUS_div_power = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L5 } add_to_variable = { KRAUS_div_power = 1 } }
	# 0.9.1：三扇门与三条第二层同记一档（一层只走一条，总档数仍是 4，buff 天数不变）
	if = { limit = { has_completed_focus = KRAUS_pl_L3b } add_to_variable = { KRAUS_div_power = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L2b } add_to_variable = { KRAUS_div_power = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L3c } add_to_variable = { KRAUS_div_power = 1 } }
	if = { limit = { has_completed_focus = KRAUS_pl_L3d } add_to_variable = { KRAUS_div_power = 1 } }
	set_variable = { KRAUS_div_buff_days = 90 }
	if = { limit = { check_variable = { KRAUS_div_power > 0 } } set_variable = { KRAUS_div_buff_days = 135 } }
}
