	else_if = { limit = { has_idea = KRAUS_jm_land_1 } set_temp_variable = { KRAUS_integ_t_mil = 1 } }

		limit = { OR = { has_idea = KRAUS_cl_loyal_2 has_idea = KRAUS_cl_loyal_3 } }
		multiply_temp_variable = { KRAUS_integ_t_econ = 1.5 }
		multiply_temp_variable = { KRAUS_integ_t_mil = 1.5 }
	}
	ROOT = {
		add_to_variable = { KRAUS_integ_div_econ = KRAUS_integ_t_econ }
		add_to_variable = { KRAUS_integ_div_mil = KRAUS_integ_t_mil }
		set_variable = { KRAUS_integ_div_ic = KRAUS_integ_div_econ }
		multiply_variable = { KRAUS_integ_div_ic = 0.01 }
		set_variable = { KRAUS_integ_div_rs = KRAUS_integ_div_econ }
		multiply_variable = { KRAUS_integ_div_rs = 0.005 }
		set_variable = { KRAUS_integ_div_org = KRAUS_integ_div_mil }
		multiply_variable = { KRAUS_integ_div_org = 0.01 }
		set_variable = { KRAUS_integ_div_cons = KRAUS_integ_div_mil }
		multiply_variable = { KRAUS_integ_div_cons = 0.005 }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = KRAUS_integ_dividend_dm } } }
			add_dynamic_modifier = { modifier = KRAUS_integ_dividend_dm }
		}
	}
