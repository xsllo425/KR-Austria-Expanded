		add_to_temp_variable = { var = KRAUS_pm_size value = 1 }
	}
	if = {
		limit = { has_country_flag = KRAUS_pm_cab_ml }
		add_to_temp_variable = { var = KRAUS_pm_size value = 1 }
	}
	if = {
		limit = { has_country_flag = KRAUS_pm_cab_sd }
		add_to_temp_variable = { var = KRAUS_pm_size value = 1 }
	}
	if = {
		limit = { has_country_flag = KRAUS_pm_cab_sl }
		add_to_temp_variable = { var = KRAUS_pm_size value = 1 }
	}
	clr_country_flag = KRAUS_pm_law_single
	clr_country_flag = KRAUS_pm_law_wait
	clr_country_flag = KRAUS_pm_law_multi
	clr_country_flag = KRAUS_pm_law_paused
	if = {
		limit = { check_variable = { var = KRAUS_pm_stable_days value = 7 compare = less_than } }
		set_country_flag = KRAUS_pm_law_paused
	}
	else_if = {
		limit = { check_variable = { var = KRAUS_pm_size value = 1 compare = less_than_or_equals } }
		set_country_flag = KRAUS_pm_law_single
		add_to_variable = { var = KRAUS_pm_law value = 2.5 }
	}
	else_if = {
		limit = { check_variable = { var = KRAUS_pm_stable_days value = 30 compare = less_than } }
		set_country_flag = KRAUS_pm_law_wait
	}
	else = {
		set_country_flag = KRAUS_pm_law_multi
		add_to_variable = { var = KRAUS_pm_law value = KRAUS_pm_size }
	}
	add_to_variable = { var = KRAUS_pm_stable_days value = 7 }
	clamp_variable = { var = KRAUS_pm_law min = 0 max = 100 }
	KRAUS_pm_law_frame_update = yes
	if = {
		limit = {
			check_variable = { var = KRAUS_pm_law value = 100 compare = greater_than_or_equals }
			NOT = { has_country_flag = KRAUS_pm_charter_ready }
		}
		set_country_flag = KRAUS_pm_charter_ready
		country_event = { id = KRAUS_pm_events.2 days = 1 }
	}
}

KRAUS_pm_activate = {
	set_variable = { var = KRAUS_pm_x value = 50 }
	set_variable = { var = KRAUS_pm_y value = 50 }
	if = {
		limit = { has_power_balance = { id = KRAUS_regime_balance } }
		if = {
			limit = { power_balance_value = { id = KRAUS_regime_balance value < -0.6 } }
			add_to_variable = { var = KRAUS_pm_x value = -10 }
			add_to_variable = { var = KRAUS_pm_y value = 15 }
		}
		else_if = {
			limit = { power_balance_value = { id = KRAUS_regime_balance value < -0.2 } }
			add_to_variable = { var = KRAUS_pm_x value = -5 }
			add_to_variable = { var = KRAUS_pm_y value = 8 }
		}
		else_if = {
			limit = { power_balance_value = { id = KRAUS_regime_balance value > 0.6 } }
			add_to_variable = { var = KRAUS_pm_y value = -15 }
		}
		else_if = {
			limit = { power_balance_value = { id = KRAUS_regime_balance value > 0.2 } }
			add_to_variable = { var = KRAUS_pm_y value = -8 }
		}
		remove_power_balance = { id = KRAUS_regime_balance }
	}
	if = {
		limit = { has_country_flag = KRAUS_dc_had_war }
		add_to_variable = { var = KRAUS_pm_y value = -5 }
	}
	else = {
		add_to_variable = { var = KRAUS_pm_y value = 5 }
	}
	set_country_flag = KRAUS_pm_active
	KRAUS_pm_recalc = yes
}
