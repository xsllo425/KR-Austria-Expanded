### 奥地利的根深蒂固问题（2026-10-11 第二版）
###
### 仿 KR 印度 india_recovery：每条减益 6 级，每次缓解直接替换为下一级精神，
### 提示里显示的就是「以 X 取代 Y」和数值变化。
###   经济危机：最后一级需黑色星期一已经过去
###   政治动荡：最后一级需皇权与秩序 ≥ 50
###   军事弱势：无额外条件
### 条件未满足时停在最后一级，条件满足后由每日检查自动移除。
### 三条全部解决且完成「看不见的手」后，发放「王冠领地的和谐发展」。
############################################################

KRAUS_ease_economic_crisis = {
	if = {
		limit = { has_idea = aus_economic_crisis }
		swap_ideas = { remove_idea = aus_economic_crisis add_idea = aus_economic_crisis_1h }
	}
	else_if = {
		limit = { has_idea = aus_economic_crisis_1h }
		swap_ideas = { remove_idea = aus_economic_crisis_1h add_idea = aus_economic_crisis_2 }
	}
	else_if = {
		limit = { has_idea = aus_economic_crisis_2 }
		swap_ideas = { remove_idea = aus_economic_crisis_2 add_idea = aus_economic_crisis_2h }
	}
	else_if = {
		limit = { has_idea = aus_economic_crisis_2h }
		swap_ideas = { remove_idea = aus_economic_crisis_2h add_idea = aus_economic_crisis_3 }
	}
	else_if = {
		limit = { has_idea = aus_economic_crisis_3 }
		swap_ideas = { remove_idea = aus_economic_crisis_3 add_idea = aus_economic_crisis_3h }
	}
	else_if = {
		limit = { has_idea = aus_economic_crisis_3h NOT = { KRAUS_has_black_monday_trigger = yes } }
		remove_ideas = aus_economic_crisis_3h
		country_event = { id = KRAUS_root_events.1 days = 1 }
	}
	else_if = {
		limit = { has_idea = aus_economic_crisis_3h }
		custom_effect_tooltip = KRAUS_root_ecr_gate_tt
		set_country_flag = KRAUS_root_ecr_ready
	}
}

# 每日：收尾条件后来才满足时自动移除；三条全清时发放终态精神
KRAUS_root_problems_update = {
	if = {
		limit = { has_country_flag = KRAUS_root_ecr_ready has_idea = aus_economic_crisis_3h NOT = { KRAUS_has_black_monday_trigger = yes } }
		clr_country_flag = KRAUS_root_ecr_ready
		remove_ideas = aus_economic_crisis_3h
		country_event = { id = KRAUS_root_events.1 days = 1 }
	}
