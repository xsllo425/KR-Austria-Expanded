############################################################
### KRAUS 1936 大选 · 风味事件组
###
### 设计依据（用户要求 2026-09-21）：
###   「可以在 1936 年大选触发前，新增多个风味事件来影响两个
###     政党的支持度，来决定最终上台的是哪个政党。」
###
### ★★ 2026-09-21 第二轮修正（用户实测反馈）★★
###   用户反馈：「事件的效果，应该在选项中显示出来，
###              基于游戏机制，而不是文字叙述。」
###
###   ★ 根因：HOI4 的 option 里直接写 add_popularity 时，
###     只有【单选项】事件会自动生成 tooltip；
###     【多选项】事件不会 —— 玩家看不到各选项的效果。
###
###   ★ 正确写法（KR 实证，00 League Collapse events.txt 等）：
###       custom_override_tooltip = {
###           tooltip = <loc_key>          # 说明文字
###           add_popularity = { ... }     # ★ 真实效果写在这里
###       }
###     effect 与 tooltip 绑在一起，玩家在按钮上就能看到。
###
###   ★ 本文件所有 option 一律采用此写法。
###
### ★ 范本：KR 原版 austria_election_events.2~.6
###   · .2 伦纳演讲 / .3 维克多·阿德勒
###   · .4 ★ 奥托王子打猎（GFX_report_event_AUS_CS_Hunt）
###   · .5 保守派得利 / .6 奥托受伤
###
### ★ 我们的做法：不覆盖 KR 原版（同 namespace 覆盖风险高），
###   改用【自己的 namespace】，在选举前触发。
###
### ★ 不使用 replace_path（三条铁律之二）
############################################################

add_namespace = KRAUS_election_flavor

############################################################
### .1 帝国新闻：黑色星期一的街头
###
### ★ 第三轮：也改为可见事件（与 .2 一致，效果显示在按钮上）
############################################################
country_event = {
	id = KRAUS_election_flavor.1
	title = KRAUS_election_flavor.1.t
	desc = KRAUS_election_flavor.1.d
	picture = GFX_report_event_generic_parliament

	is_triggered_only = yes

	option = {
		name = KRAUS_election_flavor.1.a
		log = "[GetLogInfo]: KRAUS election flavor .1 - the streets"
		# 危机中工人转向社民党 → 社民 +3%
		custom_override_tooltip = {
			tooltip = KRAUS_ef1_a_tt
			add_popularity = {
				ideology = social_democrat
				popularity = 0.03
			}
		}
	}
}

############################################################
### .2 教会的声音
###
### ★ 2026-09-21 第三轮：由「隐藏事件」改为【可见事件 + 一个按钮】
###   用户要求：效果要在选项上显示出来（像伦纳的演说那样）。
###   隐藏事件没有按钮 → 玩家看不到效果。
###   → 改为可见事件，单个 option + custom_override_tooltip。
