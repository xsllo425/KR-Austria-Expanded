### 战后秩序 · 全球去殖民化浪潮（2026-09-24）
###
### ★ 用户定案：
###   「其实这个殖民地独立的机制就是**奥地利推动大国在全球范围内，
###     让所有国家殖民地都独立的去殖民化浪潮**。因为奥地利没有殖民地，
###     而且也是以**平等博爱、团结合作、民主自由**为意识形态旗帜的一部分。」
###
### ★ 一次事件链，不做「一个个点」的决议。
###
### ★ 已核实的机制（全部来自 KR / 原版）：
###   · set_autonomy = { target = X  autonomy_state = autonomy_free }
###       出处：KR_fight_for_independence（00 Generic decisions.txt）
###   · AUS = { add_to_faction = PREV }
###       出处：KR\common\scripted_effects\00_Post_War_effects.txt:581（奥地利把主题国拉入阵营）
###   · every_subject_country / every_country / country_exists
###       KR 全库常用
###   · 殖民自治档次：colonial_government / french_colonial_government
###   · KR 认定的殖民 tag：MAF（中央非洲）/ GEA（德属东亚）/ DEI（荷属东印度）
###       is_colonial_tag = { OR = { original_tag = MAF  original_tag = GEA  original_tag = DEI } }
###
### ★ 不使用 replace_path（工程铁律）
############################################################


############################################################
### .1 去殖民化浪潮（奥地利国内事件 · 开启链条）
###   触发点：KRAUS_war_order_britain / _france / _russia 任一完成后，
###           由 KRAUS_on_actions 或国策 completion_reward 调用
############################################################
country_event = {
	id = KRAUS_war_order_events.1
	title = KRAUS_war_order_events.1.t
	desc = KRAUS_war_order_events.1.d
	picture = GFX_report_event_generic_sign_treaty2

	is_triggered_only = yes
	fire_only_once = yes

	# ★ 只有奥地利、且二战结束、且尚未发起过
	trigger = {
		original_tag = AUS
		has_global_flag = second_weltkrieg_has_ended
		NOT = { has_country_flag = KRAUS_decolonisation_launched }
	}

	# 玩家选择如何推动这场浪潮
	option = {
		name = KRAUS_war_order_events.1.a
		log = "[GetLogInfo]: event KRAUS_war_order_events.1 option a"
		ai_chance = { base = 80 }
		# 强硬推动：付出外交代价，换来更大范围
		add_political_power = -50
		add_stability = -0.03
		add_war_support = -0.02
		set_country_flag = KRAUS_decolonisation_launched
		set_country_flag = KRAUS_decolonisation_firm
		# 全球通报
		every_country = {
			limit = {
				country_exists = yes
				NOT = { tag = AUS }
			}
			country_event = { id = KRAUS_war_order_events.2 days = 3 }
		}
	}

	option = {
		name = KRAUS_war_order_events.1.b
		log = "[GetLogInfo]: event KRAUS_war_order_events.1 option b"
		ai_chance = { base = 20 }
		# 稳健推动：只对已战败的四国施压，代价小、范围小
		add_political_power = -20
		set_country_flag = KRAUS_decolonisation_launched
		set_country_flag = KRAUS_decolonisation_soft
		every_country = {
			limit = {
				country_exists = yes
				NOT = { tag = AUS }
			}
			country_event = { id = KRAUS_war_order_events.2 days = 3 }
		}
	}
}

### .2 回应（任何国家都收到 · 只有被点名的殖民宗主国会有实质选项）
############################################################
country_event = {
	id = KRAUS_war_order_events.2
	title = KRAUS_war_order_events.2.t
	desc = KRAUS_war_order_events.2.d
	picture = GFX_report_event_generic_diplomacy3

	is_triggered_only = yes

	# ★ 只有这四家是「被要求释放殖民地」的对象
	trigger = {
		OR = {
			tag = GBR
			tag = FRA
			tag = GER
			tag = POR
			# 战败政权的残余挂名也算
			original_tag = GBR
			original_tag = FRA
		}
	}

	immediate = {
		log = "[GetLogInfo]: event KRAUS_war_order_events.2 fired for [GetLogRoot]"
	}

	# a. 接受：释放全部殖民地，并让它们加入欧洲联盟
	option = {
		name = KRAUS_war_order_events.2.a
		log = "[GetLogInfo]: event KRAUS_war_order_events.2 option a"
		ai_chance = {
			base = 60
			modifier = {
				factor = 2
				has_country_flag = KRAUS_decolonisation_firm
			}
			modifier = {
				factor = 0.5
				is_subject = no
				has_war = no
			}
		}
		# ★ 释放所有殖民自治档次的主题国
		every_subject_country = {
			limit = {
				OR = {
					has_autonomy_state = colonial_government
					has_autonomy_state = french_colonial_government
					has_autonomy_state = federated_colony
				}
			}
			ROOT = {
				set_autonomy = {
					target = PREV
					autonomy_state = autonomy_free
				}
			}
			# 独立后加入欧洲联盟
			AUS = { add_to_faction = PREV }
			set_country_flag = KRAUS_was_colony_released
		}
		# 已知的殖民实体（可能不是主题国，逐个兜底）
		if = {
			limit = { country_exists = MAF }
			MAF = {
				if = { limit = { is_subject = yes } ROOT = { set_autonomy = { target = PREV autonomy_state = autonomy_free } } }
				AUS = { add_to_faction = PREV }
				set_country_flag = KRAUS_was_colony_released
			}
		}
		if = {
			limit = { country_exists = GEA }
			GEA = {
				if = { limit = { is_subject = yes } ROOT = { set_autonomy = { target = PREV autonomy_state = autonomy_free } } }
				AUS = { add_to_faction = PREV }
				set_country_flag = KRAUS_was_colony_released
			}
		}
		if = {
			limit = { country_exists = DEI }
			DEI = {
				if = { limit = { is_subject = yes } ROOT = { set_autonomy = { target = PREV autonomy_state = autonomy_free } } }
				AUS = { add_to_faction = PREV }
				set_country_flag = KRAUS_was_colony_released
			}
		}
		# 葡萄牙的非洲与亚洲殖民地（KR 认定的殖民核心：
		#   安哥拉 ANG / 莫桑比克 MZB / 几内亚比绍 GNB 为核心；
		#   果阿 321 / 佛得角 702 / 圣多美 705 / 东帝汶 721 / 澳门 729 为州）
		# ★ 用 release_autonomy + autonomy_state = autonomy_free 释放为完全独立
		#   （定义见原版 effects_documentation.md:5733；release_puppet = TAG 见 KR\LIB focus (Liberia).txt:2862）
		if = {
			limit = { country_exists = ANG }
			release = ANG
			set_autonomy = { target = ANG autonomy_state = autonomy_free }
			ANG = { set_country_flag = KRAUS_was_colony_released }
			AUS = { add_to_faction = ANG }
		}
		if = {
			limit = { country_exists = MZB }
			release = MZB
			set_autonomy = { target = MZB autonomy_state = autonomy_free }
			MZB = { set_country_flag = KRAUS_was_colony_released }
			AUS = { add_to_faction = MZB }
		}
		if = {
			limit = { country_exists = GNB }
			release = GNB
			set_autonomy = { target = GNB autonomy_state = autonomy_free }
			GNB = { set_country_flag = KRAUS_was_colony_released }
			AUS = { add_to_faction = GNB }
		}
		set_country_flag = KRAUS_accepted_decolonisation
		# 好感代价
		AUS = {
			reverse_add_opinion_modifier = { target = ROOT modifier = KR_pleased }
		}
		# ★ 通报奥地利（.3 的触发条件依赖这个）
		AUS = { country_event = { id = KRAUS_war_order_events.3 days = 5 } }
		# ★ 德国另有一摊：殖民实体（MAF / GEA）单独处理
		if = {
			limit = { tag = GER }
			country_event = { id = KRAUS_war_order_events.4 days = 10 }
		}
	}

	# b. 拒绝：奥地利施压，关系恶化，但殖民地暂不动
	option = {
		name = KRAUS_war_order_events.2.b
		log = "[GetLogInfo]: event KRAUS_war_order_events.2 option b"
		ai_chance = {
			base = 40
			modifier = {
				factor = 3
				has_country_flag = KRAUS_decolonisation_soft
			}
		}
		set_country_flag = KRAUS_refused_decolonisation
		AUS = {
			add_opinion_modifier = { target = ROOT modifier = KR_displeased }
		}
		# 奥地利记下这一笔，后续可用（例如拒绝的国家不被拉进欧盟）
		AUS = { country_event = { id = KRAUS_war_order_events.3 days = 5 } }
	}
}

### .10 奥地利提出调停（向交战双方分别发送）
###
### ★ 用户定案：
###   「一般情况下，战争状态的国家不会接受调停，但是在战争处于不利状态的一方，
###    更容易接受调停，而处于优势的一方通常不会接受。
###    而玩家可以选择加入其中任意一方。」
###
### ★ 接受率的设计（照用户原话）：
###   · 基础很低（战争中的国家一般不听劝）
###   · 不利的一方（投降度 > 0.15）→ 显著提高
###   · 优势的一方（投降度 < 0.05）→ 进一步降低
###   · 与奥地利关系好 → 略提高
###   · 已被打残（人力见底）→ 提高
############################################################
country_event = {
	id = KRAUS_war_order_events.10
	title = KRAUS_war_order_events.10.t
	desc = KRAUS_war_order_events.10.d
	picture = GFX_report_event_generic_diplomacy3

	is_triggered_only = yes

	immediate = {
		log = "[GetLogInfo]: event KRAUS_war_order_events.10 for [GetLogRoot]"
	}

	# a. 接受调停 —— 停火，按当前控制线和平
	option = {
		name = KRAUS_war_order_events.10.a
		log = "[GetLogInfo]: event KRAUS_war_order_events.10 option a"

		ai_chance = {
			base = 10
			# ★ 处于不利状态的一方更容易接受
			modifier = {
				factor = 5
				surrender_progress > 0.15
			}
			modifier = {
				factor = 3
				surrender_progress > 0.35
			}
			# ★ 处于优势的一方通常不会接受
			modifier = {
				factor = 0.3
				surrender_progress < 0.05
			}
			# 被奥地利拉过好感的，给个面子
			modifier = {
				factor = 1.5
				has_opinion = { target = AUS  value > 25 }
			}
			# 人力见底 = 打不动了
			modifier = {
				factor = 2
				has_army_manpower = { size < 50000 }
			}
		}

		set_country_flag = KRAUS_mediation_accepted
		# 通报奥地利
		AUS = { country_event = { id = KRAUS_war_order_events.11 days = 2 } }
	}

	# b. 拒绝调停
	option = {
		name = KRAUS_war_order_events.10.b
		log = "[GetLogInfo]: event KRAUS_war_order_events.10 option b"

		ai_chance = {
			base = 90
			modifier = {
				factor = 0.2
				surrender_progress > 0.15
			}
			modifier = {
				factor = 2
				surrender_progress < 0.05
			}
		}

		set_country_flag = KRAUS_mediation_refused
		AUS = { country_event = { id = KRAUS_war_order_events.12 days = 2 } }
	}
}
