# 二战中同时与德、法、俄任意两国开战：未整合的王冠领叛乱
country_event = {
	id = KRAUS_rel_events.2
	title = KRAUS_rel_events.2.t
	desc = KRAUS_rel_events.2.d
	picture = GFX_report_event_KRAUS_rel_events_2
	is_triggered_only = yes
	option = {
		name = KRAUS_rel_events.2.a
		log = "[GetLogInfo]: event KRAUS_rel_events.2 option KRAUS_rel_events.2.a"
		hidden_effect = {
			every_subject_country = {
				limit = { OR = { tag = CZE tag = GAL tag = CRO tag = HUN tag = SLO tag = TRS } }
				set_country_flag = KRAUS_cl_in_revolt
				remove_puppet_and_leave_faction = yes
				country_event = { id = KRAUS_rel_events.3 days = 1 }
			}
		}
	}
}
