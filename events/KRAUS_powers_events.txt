# 哥本哈根会议：奥地利已有协约国联络，可以做调停者（ROOT = 奥地利）
country_event = {
	id = KRAUS_pwr_events.10
	title = KRAUS_pwr_events.10.t
	desc = KRAUS_pwr_events.10.d
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes
	option = {
		name = KRAUS_pwr_events.10.a
		log = "[GetLogInfo]: event KRAUS_pwr_events.10 option KRAUS_pwr_events.10.a"
		trigger = { has_political_power > 24 }
		ai_chance = { base = 70 }
		add_political_power = -25
		add_named_threat = {
			threat = -4
			name = KRAUS_copenhagen_mediation
		}
		if = {
			limit = { country_exists = GER }
			GER = { add_opinion_modifier = { target = ROOT modifier = KRAUS_dip_friendship } }
		}
		if = {
			limit = { country_exists = ENT }
			ENT = { add_opinion_modifier = { target = ROOT modifier = KRAUS_dip_friendship } }
		}
	}
	option = {
		name = KRAUS_pwr_events.10.b
		log = "[GetLogInfo]: event KRAUS_pwr_events.10 option KRAUS_pwr_events.10.b"
		ai_chance = { base = 30 }
		add_political_power = 10
	}
}

# 第三国际密约被柏林察觉（ROOT = 奥地利）
country_event = {
	id = KRAUS_pwr_events.20
	title = KRAUS_pwr_events.20.t
	desc = KRAUS_pwr_events.20.d
	picture = GFX_report_event_generic_diplomacy3
	is_triggered_only = yes
	trigger = {
		has_idea = KRAUS_pwr_internationale_contact
		country_exists = GER
	}
	option = {
		name = KRAUS_pwr_events.20.a
		log = "[GetLogInfo]: event KRAUS_pwr_events.20 option KRAUS_pwr_events.20.a"
		ai_chance = { base = 50 }
		random_list = {
			40 = {
				add_political_power = -10
			}
			60 = {
				GER = { add_opinion_modifier = { target = ROOT modifier = KRAUS_pwr_betrayal } }
				if = {
					limit = { has_idea = KRAUS_pwr_ger_trade }
					remove_ideas = KRAUS_pwr_ger_trade
				}
			}
		}
	}
	option = {
		name = KRAUS_pwr_events.20.b
		log = "[GetLogInfo]: event KRAUS_pwr_events.20 option KRAUS_pwr_events.20.b"
		ai_chance = { base = 50 }
		add_stability = -0.02
		GER = { add_opinion_modifier = { target = ROOT modifier = KRAUS_pwr_betrayal } }
		if = {
			limit = { has_idea = KRAUS_pwr_ger_trade }
			remove_ideas = KRAUS_pwr_ger_trade
		}
	}
}
