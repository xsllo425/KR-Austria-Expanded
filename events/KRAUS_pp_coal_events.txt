# 0.9 实测 31：入阁党的加入由事件决定（基社党团）
country_event = {
	id = KRAUS_pp_coal.1
	title = KRAUS_pp_coal.1.t
	desc = KRAUS_pp_coal.1.d
	picture = GFX_report_event_KRAUS_pp_coal_1
	is_triggered_only = yes
	trigger = {
		has_country_flag = KRAUS_pm_cab_cs
		NOT = { has_country_flag = KRAUS_pm_lead_cs }
		NOT = { has_country_flag = KRAUS_pp_joined_cs }
		NOT = { has_country_flag = KRAUS_war_cabinet }
	}
	option = {
		name = KRAUS_pp_coal.1.a
		log = "[GetLogInfo]: event KRAUS_pp_coal.1 option KRAUS_pp_coal.1.a"
		set_country_flag = KRAUS_pp_joined_cs
		set_temp_variable = { coalition_partner_var = token:social_conservative }
		add_to_coalition = yes
		custom_effect_tooltip = KRAUS_pp_coal_join_cs_tt
		add_political_power = 15
	}
	option = {
		name = KRAUS_pp_coal.1.b
		log = "[GetLogInfo]: event KRAUS_pp_coal.1 option KRAUS_pp_coal.1.b"
	}
}
