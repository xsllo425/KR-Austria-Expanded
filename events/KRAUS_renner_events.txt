### .10 ★ 伦纳就任：激活平衡条（BOP 的接线点）

############################################################
country_event = {
	id = KRAUS_renner_events.10
	title = KRAUS_renner_events.10.t
	desc = KRAUS_renner_events.10.d
	picture = GFX_report_event_generic_cabinet_meeting

	is_triggered_only = yes

	option = {
		name = KRAUS_renner_events.10.a
		log = "[GetLogInfo]: KRAUS Renner - power balance activated"

		# ★ 激活帝国平衡条
		set_power_balance = {
			id = KRAUS_regime_balance
			left_side = KRAUS_renner_side
			right_side = KRAUS_miklas_side
		}
		set_country_flag = KRAUS_renner_line_active
		add_political_power = 50
	}
}
