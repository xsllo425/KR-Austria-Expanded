# ============================================================
# 撤走奥地利资产 —— 选择由谁承担
#   选项 a：让保加利亚出政治点，分它一个民工厂
#   选项 b：奥地利自己承担，两个民工厂都归自己
# ============================================================
country_event = {
	id = KRAUS_balkan_events.50
	title = KRAUS_balkan_events.50.t
	desc = KRAUS_balkan_events.50.d
	picture = GFX_report_event_generic_cabinet_meeting
	is_triggered_only = yes

	option = {
		name = KRAUS_balkan_events.50.a
		log = "[GetLogInfo]: event KRAUS_balkan_events.50 option a"
		ai_chance = { base = 50 }
		# 与保加利亚分摊：它出 50 政治点，换走 1 座；帝国留下 1 座。
		# 保加利亚拿不出政治点、或已经不存在时，两座都归帝国（走 else）。
		if = {
			limit = {
				country_exists = BUL
				BUL = { has_political_power > 49 }
			}
			BUL = {
				add_political_power = -50
				add_one_random_civilian_factory = yes
			}
			add_one_random_civilian_factory = yes
		}
		else = {
			add_one_random_civilian_factory = yes
			add_one_random_civilian_factory = yes
		}
	}

	option = {
		name = KRAUS_balkan_events.50.b
		log = "[GetLogInfo]: event KRAUS_balkan_events.50 option b"
		ai_chance = { base = 50 }
		# 帝国独吞：两座民工厂全部归自己。
		# 代价是保加利亚什么也没拿到 —— 关系会变差（见下方好感修正）。
		add_one_random_civilian_factory = yes
		add_one_random_civilian_factory = yes
		if = {
			limit = { country_exists = BUL }
			BUL = {
				reverse_add_opinion_modifier = {
					target = ROOT
					modifier = KR_displeased
				}
			}
		}
	}
}
