country_event = {
	id = KRAUS_danube_crisis.1
	title = KRAUS_danube_crisis.1.t
	desc = KRAUS_danube_crisis.1.d
	picture = GFX_report_event_KRAUS_danube_crisis_1
	is_triggered_only = yes
	option = {
		name = KRAUS_danube_crisis.1.a
		HUN = { set_rule = { can_not_declare_war = yes } }
		custom_effect_tooltip = KRAUS_dc_hun_no_declare_tt
		if = {
			limit = { has_country_flag = KRAUS_dc_won_CZE country_exists = CZE }
			if = {
				limit = { NOT = { CZE = { is_subject_of = ROOT } } }
				set_autonomy = { target = CZE autonomy_state = crownland end_wars = no }
				if = { limit = { is_faction_leader = yes } add_to_faction = CZE }
				CZE = { clr_country_flag = KRAUS_dc_released }
			}
			CZE = {
				add_to_war = {
					targeted_alliance = ROOT
					enemy = HUN
				}
				hidden_effect = { add_civil_war_target = HUN }
			}
		}
		if = {
			limit = { has_country_flag = KRAUS_dc_won_GAL country_exists = GAL }
			if = {
				limit = { NOT = { GAL = { is_subject_of = ROOT } } }
				set_autonomy = { target = GAL autonomy_state = crownland end_wars = no }
				if = { limit = { is_faction_leader = yes } add_to_faction = GAL }
				GAL = { clr_country_flag = KRAUS_dc_released }
			}
			GAL = {
				add_to_war = {
					targeted_alliance = ROOT
					enemy = HUN
				}
				hidden_effect = { add_civil_war_target = HUN }
			}
		}
		if = {
			limit = { has_country_flag = KRAUS_dc_won_CRO country_exists = CRO }
			if = {
				limit = { NOT = { CRO = { is_subject_of = ROOT } } }
				set_autonomy = { target = CRO autonomy_state = crownland end_wars = no }
				if = { limit = { is_faction_leader = yes } add_to_faction = CRO }
				CRO = { clr_country_flag = KRAUS_dc_released }
			}
			CRO = {
				add_to_war = {
					targeted_alliance = ROOT
					enemy = HUN
				}
				hidden_effect = { add_civil_war_target = HUN }
			}
		}
		if = {
			limit = {
				NOT = { has_country_flag = KRAUS_dc_won_CZE }
				NOT = { has_country_flag = KRAUS_dc_won_GAL }
				NOT = { has_country_flag = KRAUS_dc_won_CRO }
			}
			custom_effect_tooltip = KRAUS_dc_none_join_tt
		}
	}
}
